Adds a progress bar under the heading showing the share of tasks that are done (0 when there are none).

```ts
// "Tasks" main layer: starred tasks first, a filter box, delete on each task, and a progress bar.
import { col, row, heading, label, button, input, toggle, el, attr, end,
         getNum, setNum, getStr, setStr, eventId, cap } from "./llp";

// v2 added tasks.<i>.star (0 or 1). The filter text lives in "filter" and needs no migration.
export function llp_schema(): i32 { return 2; }
export function llp_migrate(from: i32): void {
  if (from < 2) {
    const n = <i32>getNum("tasks.n");
    for (let i = 0; i < n; i++) setNum("tasks." + i.toString() + ".star", 0);
  }
}

function key(i: i32): string { return "tasks." + i.toString(); }
function starred(i: i32): bool { return getNum(key(i) + ".star") > 0; }
function matches(i: i32, q: string): bool {
  return q.length == 0 || getStr(key(i) + ".text").toLowerCase().includes(q);
}

function drawTask(i: i32): void {
  const k = key(i);
  row();
    button("s:" + i.toString(), starred(i) ? "★" : "☆");
    toggle("t:" + i.toString(), getNum(k + ".done") > 0, getStr(k + ".text"));
    button("d:" + i.toString(), "×", "danger");
  end();
}

// copy task `from` into slot `to`
function move(from: i32, to: i32): void {
  setStr(key(to) + ".text", getStr(key(from) + ".text"));
  setNum(key(to) + ".done", getNum(key(from) + ".done"));
  setNum(key(to) + ".star", getNum(key(from) + ".star"));
}

function progress(pct: i32): void { el("progress"); attr("value", pct.toString()); end(); }

export function llp_view(): void {
  const n = <i32>getNum("tasks.n");
  const q = getStr("filter").toLowerCase();
  let done = 0, stars = 0, shown = 0;
  for (let i = 0; i < n; i++) {
    if (getNum(key(i) + ".done") > 0) done++;
    if (starred(i)) stars++;
    if (matches(i, q)) shown++;
  }

  col();
    heading("Tasks");
    label(done.toString() + " of " + n.toString() + " done · " + stars.toString() + " starred");
    progress(n == 0 ? 0 : done * 100 / n);
    input("filter", "Filter tasks…");
    row();
      input("draft", "Add a task…", "add");
      button("add", "Add", "primary");
    end();
    // starred first, then the rest, each group in the order they were added
    for (let i = 0; i < n; i++) if (starred(i) && matches(i, q)) drawTask(i);
    for (let i = 0; i < n; i++) if (!starred(i) && matches(i, q)) drawTask(i);
    if (q.length > 0 && shown < n) label((n - shown).toString() + " hidden by the filter");
    if (done > 0) button("clear", "Clear done");
  end();
}

export function llp_event(): void {
  const id = eventId();
  if (id == "add") {
    const draft = getStr("draft");
    if (draft.length == 0) return;
    const n = <i32>getNum("tasks.n");
    setStr(key(n) + ".text", draft);
    setNum(key(n) + ".done", 0);
    setNum(key(n) + ".star", 0);
    setNum("tasks.n", n + 1);
    setStr("draft", "");
  } else if (id.startsWith("t:")) {
    const k = "tasks." + id.substr(2);
    const now = getNum(k + ".done") > 0 ? 0.0 : 1.0;
    setNum(k + ".done", now);
    if (now > 0) cap("haptic", "20");
  } else if (id.startsWith("s:")) {
    const k = "tasks." + id.substr(2);
    setNum(k + ".star", getNum(k + ".star") > 0 ? 0.0 : 1.0);
  } else if (id.startsWith("d:")) {
    // delete one task: shift the ones after it down by one
    const n = <i32>getNum("tasks.n");
    const at = I32.parseInt(id.substr(2));
    if (at < 0 || at >= n) return;
    for (let i = at; i < n - 1; i++) move(i + 1, i);
    setNum("tasks.n", n - 1);
  } else if (id == "clear") {
    // keep the undone tasks, moving text and star together
    const n = <i32>getNum("tasks.n");
    let w = 0;
    for (let i = 0; i < n; i++) {
      if (getNum(key(i) + ".done") == 0) { if (w != i) move(i, w); w++; }
    }
    setNum("tasks.n", w);
  }
}
```
